  7async function authenticate(req, res, next) {
  8  const match = /^Bearer (\S+)$/.exec(req.get('Authorization') || '');
  9  if (!match) return res.status(401).json({ message: 'No autorizado' });
 10
 11  let payload;
 12  try {
 13    payload = jwt.verify(match[1], env.JWT_SECRET, { algorithms: ['HS256'] });
 14  } catch (error) {
 15    if (error instanceof jwt.JsonWebTokenError || error instanceof jwt.TokenExpiredError) {
 16      return res.status(401).json({ message: 'No autorizado' });
 17    }
 18    throw error;
 19  }
 20
 21  if (
 22    typeof payload !== 'object' ||
 23    !z.uuid().safeParse(payload.sub).success ||
 24    !Number.isSafeInteger(payload.tokenVersion) || payload.tokenVersion < 0 ||
 25    typeof payload.exp !== 'number' ||
 26    payload.exp <= Math.floor(Date.now() / 1000)
 27  ) {
 28    return res.status(401).json({ message: 'No autorizado' });
 29  }
 30
 31  // La base es la fuente actual del rol, estado y versión de sesión.
 32  const user = await prisma.user.findUnique({
 33    where: { id: payload.sub },
 34    select: { ...publicUserSelect, isActive: true, tokenVersion: true },
 35  });
 36  if (!user || !user.isActive || user.tokenVersion !== payload.tokenVersion) {
 37    return res.status(401).json({ message: 'No autorizado' });
 38  }
 39
 40  const { isActive, tokenVersion, ...safeUser } = user;
 41  req.user = safeUser;
 42  req.tokenVersion = tokenVersion;
 43  return next();
 44}

 46function authorizeRoles(...roles) {
 47  return (req, res, next) => {
 48    if (!req.user) return res.status(401).json({ message: 'No autorizado' });
 49    if (!roles.includes(req.user.role)) return res.status(403).json({ message: 'Prohibido' });
 50    return next();
 51  };
 52}

 11  return async function request(path, { method = 'GET', body, params } = {}) {
 12    const query = new URLSearchParams();
 13    Object.entries(params || {}).forEach(([key, value]) => {
 14      if (value !== '' && value !== undefined && value !== null) query.set(key, String(value));
 15    });
 16    const token = getToken();
 17    let response;
 18    try {
 19      response = await fetch(`${baseUrl}${path}${query.size ? `?${query}` : ''}`, {
 20        method,
 21        headers: { ...(body !== undefined ? { 'Content-Type': 'application/json' } : {}), ...(token ? { Authorization: `Bearer ${token}` } : {}) },
 22        ...(body !== undefined ? { body: JSON.stringify(body) } : {}),
 23      });
 24    } catch {
 25      throw new ApiError(0, 'No se pudo conectar con el servidor. Inténtalo de nuevo.');
 26    }
 27    const payload = await response.json().catch(() => ({}));
 28    if (!response.ok) {
 29      if (response.status === 401 && token) onUnauthorized();
 30      const fallback = response.status === 403 ? 'No tienes permiso para esta operación.'
 31        : response.status === 409 ? 'El registro entra en conflicto con datos existentes o referencias inactivas.'
 32          : response.status === 400 ? 'Revisa los datos ingresados.' : 'Ocurrió un error. Inténtalo de nuevo.';
